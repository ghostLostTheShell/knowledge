# 可选属性的定义

```ts
type Foo = {
    a?: string
}

//或

interface Foo {
    a?: string
}
```

# keyof 的用法

```ts
interface Point {
	name: string;
	age: number;
	gender: string;
}
type Key = keyof Point // 返回的是'name' | 'age' | 'gender'
```

# 可索引类型

```ts

interface 字符串数组 {
  [index: number]: string;
}

// 定义一个字符串数组类型

let myArray: 字符串数组;

myArray = ["Bob", "Fred"];

```

# 自读可索引类型

```ts

interface ReadonlyStringArray {
    readonly [index: number]: string; //设置为自读
}

let myArray: ReadonlyStringArray = ["Alice", "Bob"];
myArray[2] = "Mallory"; // error

```
