#

## 解构赋值

例子

```js

a = {
  p1: "P1",
  p2: "p2"
}
const {p1, p2} = a

// 支持多层结构

b = {
  p3: {
        pp1: 123,
        pp2: 456
      },
      p4:789
       }


const {p3: { pp1 }, p4} = b
```

## ... 为获取剩下的

```js
var a = {
    h:1,
    b:"22",
    hh: "123",
    l:"567"
  };

let { h, b, ...other } = a;

other

// => { b: '22', hh: '123' }
  
```

## 函数形参结构

```js
function t({name="默认值", age, sex, ...${剩余参数:object}}){
  console.log(`姓名：${name}, 年龄￥${age}, 性别${sex}`)
}


t({name:"鼠标"， age: 55, sex:null})

```
